% ---------------------------------------------------------------- 30 s
\begin{frame}{Two machines, and only one of them talks to you}
  \begin{itemize}
    \item A \alert{model} predicts, learns as it goes, and \alert{repairs
          itself} when the world stops matching what it learned.
    \item A separate \alert{watchdog} reads the model's mistakes and raises
          an alarm once they pile up.
    \item The alarm is the only thing a human ever sees. It is what triggers
          the audit, the retraining, the pause.
  \end{itemize}
  \vfill
  \begin{center}
    \large Nobody watches the model. The watchdog does.
  \end{center}
  \note{Picture two machines. The first one predicts something, learns as it
    goes, and repairs itself when the world stops matching what it learned.
    The second one is a watchdog. It reads the first one's mistakes and
    raises an alarm once they pile up. That alarm is the only thing a human
    ever sees: it is what triggers the audit, the retraining, the pause.
    Nobody watches the model directly. The watchdog does.}
\end{frame}

% ---------------------------------------------------------------- 40 s
\begin{frame}{The repair destroys the evidence}
  \begin{center}
    \includegraphics[height=0.74\textheight]{Fig_slide_mecanisme.png}
  \end{center}
  \note{Here is what goes wrong. The world changes, so the model starts
    making mistakes: that is the top curve. The watchdog needs those
    mistakes. It adds them up and calls for help once the pile is high
    enough. But the model repairs itself, the mistakes stop, and the pile
    stops growing too early. The repair has erased the evidence. The better
    the model fixes itself, the more surely the alarm stays silent.}
\end{frame}

% ---------------------------------------------------------------- 35 s
\begin{frame}{It is the normal case, not a corner case}
  \vspace{0.4em}
  \begin{center}
    {\fontsize{46}{52}\selectfont \alert{\textbf{3}} alarms\\[0.35em]
     \fontsize{46}{52}\selectfont out of \alert{\textbf{2\,000}} runs}

    \vspace{0.5em}
    {\small re-running the source paper's own benchmark, here, on our machines}
  \end{center}
  \vspace{0.6em}
  \begin{itemize}
    \item One research field measures how fast a model recovers. Another
          measures how fast an alarm fires.
    \item \alert{Neither has measured them against each other.}
  \end{itemize}
  \note{This is not a corner case. We re-ran the benchmark of the paper we
    are studying, and over two thousand runs the alarm fired three times.
    Now, why had nobody caught this? Because two separate research fields
    are involved. One studies models that repair themselves and measures how
    fast they recover. The other studies alarms and measures how fast they
    fire. Neither had measured the two against each other. That gap is our
    subject.}
\end{frame}

% ---------------------------------------------------------------- 40 s
\begin{frame}{What we found: the clock was wrong}
  \begin{center}
    \includegraphics[width=\textwidth]{Fig_slide_deux_horloges.png}
  \end{center}
  \vspace{0.2em}
  \begin{center}
    \alert{\large The accepted date marks the first tree out of ten,
    not a repaired forest.}
  \end{center}
  \note{To study a race you need a stopwatch, and that is where we found
    something. The model is a forest of ten trees, and it repairs itself by
    replacing them one at a time. The date everyone uses to say it has
    recovered is the moment the very first tree is replaced: one out of ten.
    Wait instead for three quarters of the forest, and you wait roughly ten
    times longer. The accepted stopwatch does not time a repair. It times
    the start of one, always too early.}
\end{frame}

% ---------------------------------------------------------------- 25 s
\begin{frame}{Where we stand}
  \begin{itemize}
    \item The paradox is \alert{real and reproducible}, and we can now say
          exactly when the evidence falls short.
    \item The measure the field relies on is \alert{biased, always the same
          way}. Any conclusion drawn from it inherits that bias.
    \item \textbf{Next}: watch the model and the watchdog on one common
          clock, and turn this into advice a practitioner can use.
  \end{itemize}
  \vfill
  \begin{center}
    \large Thank you.
  \end{center}
  \note{Where we stand. The paradox is real and reproducible, and we can now
    say exactly when the evidence falls short of what the alarm needs. The
    measure the field relies on is biased, and always in the same direction,
    so any conclusion drawn from it inherits that bias. What remains is to
    watch the model and the watchdog on one common clock, and turn this into
    advice a practitioner can actually use. Thank you.}
\end{frame}
